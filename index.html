import math def fiber_hesapla(guc_in_dbm, uzunluk_km, kayip_db_per_km=0.2, ek_kayip_db=0): """ Fiber optik güç hesaplayıcı guc_in_dbm: Giriş gücü (dBm) uzunluk_km: Fiber uzunluğu (km) kayip_db_per_km: km başına kayıp (tipik 0.2 dB/km) """ toplam_kayip = (uzunluk_km * kayip_db_per_km) + ek_kayip_db guc_out_dbm = guc_in_dbm - toplam_kayip # mW'a çevir guc_out_mw = 10 ** (guc_out_dbm / 10) return { "toplam_kayip_db": round(toplam_kayip, 2), "cikis_gucu_dbm": round(guc_out_dbm, 2), "cikis_gucu_mw": round(guc_out_mw, 4) } # Test if __name__ == "__main__": sonuc = fiber_hesapla(guc_in_dbm=10, uzunluk_km=25) print(f"Toplam Kayıp: {sonuc['toplam_kayip_db']} dB") print(f"Çıkış Gücü: {sonuc['cikis_gucu_dbm']} dBm ({sonuc['cikis_gucu_mw']} mW)") # Uyarı if sonuc['cikis_gucu_dbm'] < -25: print("UYARI: Sinyal çok zayıf! Amplifikatör gerekebilir.") else: print("Sinyal seviyesi OK.")
